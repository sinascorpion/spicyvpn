cmake_minimum_required(VERSION 3.22)
project(spicyopenvpn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json dependency_manifest)
foreach(dep ovpn asio fmt lz4 mbedtls mbedframework)
  string(JSON repository GET "${dependency_manifest}" ${dep} repository)
  string(JSON revision GET "${dependency_manifest}" ${dep} tag)
  set(${dep}_URL "https://github.com/${repository}/archive/${revision}.tar.gz")
endforeach()
foreach(dep ovpn asio fmt lz4 mbedtls)
  FetchContent_Declare(${dep} URL ${${dep}_URL})
endforeach()
foreach(dep ovpn asio fmt lz4 mbedtls)
  FetchContent_GetProperties(${dep})
  if(NOT ${dep}_POPULATED)
    FetchContent_Populate(${dep})
  endif()
endforeach()
# OpenVPN checks certificate names/roles itself, rather than assuming a DNS CN.
# Mbed TLS 3.6.7 requires an explicit hostname call even in that case.
set(ssl_context_header ${ovpn_SOURCE_DIR}/openvpn/mbedtls/ssl/sslctx.hpp)
file(READ ${ssl_context_header} ssl_context_source)
string(REPLACE "if (hostname && ((c.flags & SSLConst::ENABLE_CLIENT_SNI) || c.ca_chain))"
  "if (!hostname || ((c.flags & SSLConst::ENABLE_CLIENT_SNI) || c.ca_chain))"
  ssl_context_patched "${ssl_context_source}")
if(NOT ssl_context_source STREQUAL ssl_context_patched)
  file(WRITE ${ssl_context_header} "${ssl_context_patched}")
endif()
FetchContent_Declare(mbedframework
  URL ${mbedframework_URL})
FetchContent_Populate(mbedframework)
file(COPY ${mbedframework_SOURCE_DIR}/ DESTINATION ${mbedtls_SOURCE_DIR}/framework)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory(${mbedtls_SOURCE_DIR} ${mbedtls_BINARY_DIR} EXCLUDE_FROM_ALL)
set(core_sources ${ovpn_SOURCE_DIR}/client/ovpncli.cpp
  ${ovpn_SOURCE_DIR}/openvpn/crypto/data_epoch.cpp ${lz4_SOURCE_DIR}/lib/lz4.c)
if(ANDROID)
  add_library(spicyopenvpn SHARED bridge.cpp ${core_sources})
else()
  add_executable(spicyopenvpn tests/session_test.cpp ${core_sources})
  target_compile_definitions(spicyopenvpn PRIVATE USE_TUN_BUILDER)
endif()
target_include_directories(spicyopenvpn PRIVATE ${ovpn_SOURCE_DIR} ${asio_SOURCE_DIR}/include
  ${fmt_SOURCE_DIR}/include ${lz4_SOURCE_DIR}/lib ${mbedtls_SOURCE_DIR}/include)
target_compile_definitions(spicyopenvpn PRIVATE USE_ASIO ASIO_STANDALONE USE_MBEDTLS HAVE_LZ4 FMT_HEADER_ONLY)
target_compile_options(spicyopenvpn PRIVATE -ffunction-sections -fdata-sections)
target_link_options(spicyopenvpn PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
target_link_libraries(spicyopenvpn PRIVATE mbedtls mbedx509 mbedcrypto)
if(ANDROID)
  target_link_libraries(spicyopenvpn PRIVATE log)
else()
  target_link_libraries(spicyopenvpn PRIVATE pthread)
endif()
